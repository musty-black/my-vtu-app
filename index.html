function approveFunding(index){ let fund = pendingFundings[index]; walletBalance += fund.amount; // FIX: Find am for History and change to Success let allHistory = JSON.parse(localStorage.getItem('transactions') || '[]'); let found = allHistory.find(t => t.amount == fund.amount && t.status == 'Pending'); if(found){ found.status = 'Success'; } localStorage.setItem('transactions', JSON.stringify(allHistory)); pendingFundings.splice(index,1); localStorage.setItem('wallet', walletBalance); location.reload(); }
